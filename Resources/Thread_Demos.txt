	private static void demo5() throws InterruptedException {
		List<String> files = List.of("report.pdf", "data.csv", "image.png", "notes.txt", "backup.zip");

		ExecutorService executor = Executors.newFixedThreadPool(3);

		for (String file : files) {
			executor.submit(() -> processFile(file));
		}

		executor.shutdown();
		executor.awaitTermination(1, TimeUnit.MINUTES);
	}
	
	private static void processFile(String fileName) {
		System.out.println(Thread.currentThread().getName() + " processing " + fileName);

		try {
			Thread.sleep(1000);
		} catch (InterruptedException e) {
			Thread.currentThread().interrupt();
		}

		System.out.println(Thread.currentThread().getName() + " finished " + fileName);
	}

	private static void demo4() {
		ExecutorService executor = Executors.newFixedThreadPool(2);

		for (int i = 0; i < 5; i++) {
			executor.submit(() -> {
				System.out.println(Thread.currentThread().getName() + " running");
			});
		}

		executor.shutdown();
	}

	private static void demo3() {
