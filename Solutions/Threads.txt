//---------- Demo1 -----------------

class Counter {
	private int value = 0;

	public void increment() {
		value++;
	}

	public int getValue() {
		return value;
	}
}


class Worker implements Runnable {
	private Counter counter;

	public Worker(Counter counter) {
		this.counter = counter;
	}

	@Override
	public void run() {
		System.out.println(Thread.currentThread().getName() + " started");

		for (int i = 0; i < 10000; i++) {
			counter.increment();
		}

		System.out.println(Thread.currentThread().getName() + " finished");
	}
}

public class ThreadMain {
	public static void main(String[] args) throws InterruptedException {
		Counter counter = new Counter();

		int threadCount = 7;

		Thread[] threads = new Thread[threadCount];

		for (int i = 0; i < threadCount; i++) {
			threads[i] = new Thread(new Worker(counter));

			threads[i].start();
		}

		for (Thread t : threads) {
			t.join();
		}

		System.out.println("\nExpected: " + (threadCount * 10000));

		System.out.println("Actual: " + counter.getValue());
	}
}

import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

public class ThreadMain2 {
	public static void main(String[] args) throws InterruptedException {
		Counter counter = new Counter();

		int taskCount = 7;

		ExecutorService executor = Executors.newFixedThreadPool(5);

		for (int i = 0; i < taskCount; i++) {
			executor.submit(new Worker(counter));
		}

		executor.shutdown();

		executor.awaitTermination(1, TimeUnit.MINUTES);

		System.out.println("\nExpected: " + (taskCount * 10000));

		System.out.println("Actual: " + counter.getValue());
	}
}

using System;
using System.Threading;
using System.Threading.Tasks;

class SafeCounter
{
    private int value = 0;
    private readonly object lockObj = new object();

    public void Increment()
    {
        lock (lockObj)   // same as synchronized in Java
        {
            value++;
        }
    }

    public int GetValue()
    {
        lock (lockObj)
        {
            return value;
        }
    }
}

class SafeWorker
{
    private readonly SafeCounter counter;

    public SafeWorker(SafeCounter counter)
    {
        this.counter = counter;
    }

    public void Run()
    {
        Console.WriteLine($"{Thread.CurrentThread.ManagedThreadId} started");

        for (int i = 0; i < 10000; i++)
        {
            counter.Increment();
        }

        Console.WriteLine($"{Thread.CurrentThread.ManagedThreadId} finished");
    }
}

class ThreadMain3
{
    static void Main()
    {
        SafeCounter counter = new SafeCounter();
        int taskCount = 7;

        Task[] tasks = new Task[taskCount];

        for (int i = 0; i < taskCount; i++)
        {
            SafeWorker worker = new SafeWorker(counter);
            tasks[i] = Task.Run(() => worker.Run());
        }

        Task.WaitAll(tasks);

        Console.WriteLine($"\nExpected: {taskCount * 10000}");
        Console.WriteLine($"Actual:   {counter.GetValue()}");
    }
}
